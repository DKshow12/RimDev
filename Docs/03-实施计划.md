# RimDev 实施计划 · 第 1 期（骨架 + 环境 + 元数据查证）

> **For agentic workers:** REQUIRED SUB-SKILL: 本计划按任务顺序执行，每个 Task 内按 `- [ ]` 复选框逐步推进；**不要跳步**，也不要合并 Task。每个 Task 末尾必须跑自测并看到期望输出，再进入下一个 Task。

**Goal:** 交付一个可离线运行的 `rimdev` CLI，能探测环世界环境、把 `Assembly-CSharp.dll` 的类型/成员反射成查询表，并回答"这个类型/字段/属性到底存不存在"——把项目历史上 8 次「猜名字」造成的加载期错误提前到命令行阶段拦截。

**Architecture:** 三个 .NET 8 项目（Core 类库 / Cli 控制台 / SelfTest 自测台），手写 csproj，**零外部 NuGet 依赖**。`AssemblyLoadContext.Default.Resolving` 把 `RimWorldWin64_Data\Managed` 作为解析根，`Assembly.LoadFrom` 载入 `Assembly-CSharp.dll` 后建立**成员索引**（字段含 NonPublic，属性，方法）；`NameOracle` 在这个索引上做存在性判定与候选建议；命令层只做参数解析与报告输出。

**Tech Stack:** C# 12 / .NET 8（`Microsoft.NETCore.App 8.0.30` 已装）、`System.Text.Json`（共享框架自带）、手写 csproj（**不能用 `dotnet new`**）、自研断言测试台。

### 0.1 三条必须写进代码的成员表规则（实测得出）

1. **字段必须枚举 `Public | NonPublic | Instance`。**
   证据：`VerbProperties.forcedMissRadius` **是 private 字段**（`IsPublic=False IsPrivate=True`），而它是本项目炮塔 Def 正在使用的合法 XML 字段。
   若只枚举 public 字段 → 会把 `forcedMissRadius` 报成"不存在"，**比不做校验更糟**。
   计数：`ThingDef` public 266 / 全部 275；`VerbProperties` 113 / 116；`RaceProperties` 114 / 120。

2. **成员查找必须大小写敏感，并给出大小写纠正建议。**
   证据：`RaceProperties.IsMechanoid`（大写 I）存在，`RaceProperties.isMechanoid`（小写）不存在。前者是属性、后者是错误写法。

3. **XML 只从字段读，规则包需要属性。**
   证据：`ThingDef.defName` / `label` / `description` 都是**字段**（属性为"无"）。
   因此：`api member` 与 `name check` 要同时覆盖字段与属性，并在报告里标注 `memberKind`。

```csharp
using System.Text;

namespace RimDev.SelfTest;

/// <summary>
/// 极简断言台。设计约束：本机 nuget.org 不可达、无 xunit / MSTest 可用，
/// 因此测试以"注册用例 + 抛异常即失败"的方式手写。
/// </summary>
public static class T
{
    private sealed record Result(string Name, bool Ok, string? Message, long Ms);

    private static readonly List<Result> Results = new();
    private static string _group = "default";

    public static void Group(string name) => _group = name;

    public static void Case(string name, Action body)
    {
        var sw = System.Diagnostics.Stopwatch.StartNew();
        try
        {
            body();
            Results.Add(new Result($"{_group}/{name}", true, null, sw.ElapsedMilliseconds));
        }
        catch (Exception ex)
        {
            Results.Add(new Result($"{_group}/{name}", false, ex.Message, sw.ElapsedMilliseconds));
        }
    }

    public static void Eq<T>(T actual, T expected, string what)
    {
        if (!EqualityComparer<T>.Default.Equals(actual, expected))
            throw new InvalidOperationException($"{what}: 期望 [{expected}]，实际 [{actual}]");
    }

    public static void True(bool value, string what)
    {
        if (!value) throw new InvalidOperationException($"{what}: 期望 true，实际 false");
    }

    public static void False(bool value, string what) => True(!value, what);

    public static void NotNull(object? value, string what)
    {
        if (value is null) throw new InvalidOperationException($"{what}: 期望非 null，实际 null");
    }

    public static void Null(object? value, string what)
    {
        if (value is not null) throw new InvalidOperationException($"{what}: 期望 null，实际 [{value}]");
    }

    public static void Contains(string haystack, string needle, string what)
    {
        if (haystack is null || !haystack.Contains(needle, StringComparison.OrdinalIgnoreCase))
            throw new InvalidOperationException($"{what}: [{haystack ?? "null"}] 中未包含 [{needle}]");
    }

    /// <summary>打印汇总，返回进程退出码（0=全过，1=有失败）。</summary>
    public static int Report()
    {
        var sb = new StringBuilder();
        int passed = 0, failed = 0;

        string? lastGroup = null;
        foreach (var r in Results)
        {
            var group = r.Name.Split('/')[0];
            if (group != lastGroup)
            {
                sb.AppendLine($"--- {group} ---");
                lastGroup = group;
            }

            if (r.Ok)
            {
                passed++;
                sb.AppendLine($"  PASS {r.Name.Split('/')[1]} ({r.Ms} ms)");
            }
            else
            {
                failed++;
                sb.AppendLine($"  FAIL {r.Name.Split('/')[1]} ({r.Ms} ms)");
                sb.AppendLine($"       {r.Message}");
            }
        }

        sb.AppendLine();
        sb.AppendLine($"合计 {Results.Count} 个用例：{passed} 通过 / {failed} 失败");
        Console.WriteLine(sb.ToString());

        if (failed > 0)
        {
            Console.WriteLine("有失败用例。失败清单：");
            foreach (var r in Results.Where(x => !x.Ok))
                Console.WriteLine($"  - {r.Name}: {r.Message}");
        }

        return failed > 0 ? 1 : 0;
    }
}
```

Create `_rimdev\engine\RimDev.Core\Environment\GameEnvProbe.cs`：

```csharp
namespace RimDev.Core.Environment;

public sealed class GameEnvProbeException : Exception
{
    public GameEnvProbeException(string message) : base(message) { }
}

/// <summary>
/// 探测环世界安装位置。顺序：环境变量 RIMWORLD_DIR → 从工作区向上找 RimWorldWin64.exe
/// → Steam 注册表 → 常见默认路径。全部失败抛 <see cref="GameEnvProbeException"/>。
/// </summary>
public static class GameEnvProbe
{
    public const int ExpectedManagedDllCount = 108;

    private static readonly string[] DefaultRoots =
    {
        @"C:\Program Files (x86)\Steam\steamapps\common\RimWorld",
        @"C:\Program Files\Steam\steamapps\common\RimWorld",
        @"D:\Steam\steamapps\common\RimWorld",
        @"D:\SteamLibrary\steamapps\common\RimWorld",
        @"E:\SteamLibrary\steamapps\common\RimWorld",
    };

    public static GameEnv Discover(string? explicitRoot = null, string? startDir = null)
    {
        var root = explicitRoot
                   ?? System.Environment.GetEnvironmentVariable("RIMWORLD_DIR")
                   ?? FindByWalkingUp(startDir ?? Directory.GetCurrentDirectory())
                   ?? FindBySteamRegistry()
                   ?? DefaultRoots.FirstOrDefault(IsGameRoot);

        if (root is null || !IsGameRoot(root))
            throw new GameEnvProbeException(
                "找不到 RimWorld 游戏根目录。请设置环境变量 RIMWORLD_DIR，或在游戏根目录下运行。");

        return FromRoot(root);
    }

    public static bool IsGameRoot(string dir) =>
        Directory.Exists(dir) && File.Exists(Path.Combine(dir, "RimWorldWin64.exe"));

    /// <summary>从 startDir 逐级上溯；工作区是 &lt;游戏根&gt;\Mods，因此上一级即命中。</summary>
    public static string? FindByWalkingUp(string startDir)
    {
        var dir = new DirectoryInfo(Path.GetFullPath(startDir));
        while (dir is not null)
        {
            if (IsGameRoot(dir.FullName)) return dir.FullName;
            dir = dir.Parent;
        }
        return null;
    }

    public static string? FindBySteamRegistry()
    {
        if (!OperatingSystem.IsWindows()) return null;
        try
        {
            var steamPath = Microsoft.Win32.Registry.GetValue(
                @"HKEY_CURRENT_USER\Software\Valve\Steam", "SteamPath", null) as string;
            if (string.IsNullOrWhiteSpace(steamPath)) return null;

            var candidate = Path.Combine(steamPath, "steamapps", "common", "RimWorld");
            if (IsGameRoot(candidate)) return candidate;

            // 库文件夹可能在别的盘
            var libraryFolders = Path.Combine(steamPath, "steamapps", "libraryfolders.vdf");
            if (!File.Exists(libraryFolders)) return null;

            foreach (var line in File.ReadLines(libraryFolders))
            {
                var t = line.Trim();
                if (!t.StartsWith("\"path\"", StringComparison.Ordinal)) continue;
                var parts = t.Split('"', StringSplitOptions.RemoveEmptyEntries);
                if (parts.Length < 3) continue;
                var p = parts[^1].Replace("\\\\", "\\");
                var c = Path.Combine(p, "steamapps", "common", "RimWorld");
                if (IsGameRoot(c)) return c;
            }
        }
        catch
        {
            // 注册表不可用属正常，继续走默认路径
        }
        return null;
    }

    public static GameEnv FromRoot(string gameRoot)
    {
        gameRoot = Path.GetFullPath(gameRoot);
        var managed = Path.Combine(gameRoot, "RimWorldWin64_Data", "Managed");
        var data = Path.Combine(gameRoot, "Data");
        var mods = Path.Combine(gameRoot, "Mods");

        var localLow = Path.Combine(
            System.Environment.GetFolderPath(System.Environment.SpecialFolder.UserProfile),
            "AppData", "LocalLow", "Ludeon Studios", "RimWorld by Ludeon Studios");
        var config = Path.Combine(localLow, "Config");

        var versionRaw = ReadTrimmed(Path.Combine(gameRoot, "Version.txt"));
        var modsConfigPath = Path.Combine(config, "ModsConfig.xml");
        var versionConfig = ReadModsConfigVersion(modsConfigPath);

        // 版本口径冲突（实测：Version.txt=rev590 而 ModsConfig=rev591）时以 ModsConfig 为准，
        // 因为它才是游戏自己写入、并用于选择 loadFolders 版本的那个值。
        var versionSource = string.IsNullOrWhiteSpace(versionConfig) ? versionRaw : versionConfig;

        return new GameEnv(
            GameRoot: gameRoot,
            ManagedDir: managed,
            ModsDir: mods,
            DataDir: data,
            WorkshopDir: ResolveWorkshopDir(gameRoot),
            ConfigDir: config,
            ModsConfigPath: modsConfigPath,
            PlayerLogPath: Path.Combine(localLow, "Player.log"),
            PrevPlayerLogPath: Path.Combine(localLow, "Player-prev.log"),
            VersionRaw: versionRaw,
            VersionConfig: versionConfig,
            TargetVersion: ToMajorMinor(versionSource),
            ManagedDllCount: Directory.Exists(managed)
                ? Directory.GetFiles(managed, "*.dll").Length
                : 0);
    }

    private static string ResolveWorkshopDir(string gameRoot)
    {
        var steamApps = Directory.GetParent(gameRoot)?.Parent?.FullName; // ...\steamapps
        if (steamApps is not null)
        {
            var candidate = Path.Combine(steamApps, "workshop", "content", "294100");
            if (Directory.Exists(candidate)) return candidate;
        }
        return Path.Combine(steamApps ?? gameRoot, "workshop", "content", "294100");
    }

    /// <summary>"1.6.4871 rev591" → "1.6"</summary>
    public static string ToMajorMinor(string version)
    {
        if (string.IsNullOrWhiteSpace(version)) return "";
        var core = version.Split(' ', StringSplitOptions.RemoveEmptyEntries)[0];
        var parts = core.Split('.');
        return parts.Length >= 2 ? parts[0] + "." + parts[1] : core;
    }

    private static string ReadTrimmed(string path) =>
        File.Exists(path) ? File.ReadAllText(path).Trim() : "";

    /// <summary>从 ModsConfig.xml 取 &lt;version&gt;，避免引 XML 解析器（此处只需一个值）。</summary>
    private static string ReadModsConfigVersion(string path)
    {
        if (!File.Exists(path)) return "";
        try
        {
            var text = File.ReadAllText(path);
            var open = text.IndexOf("<version>", StringComparison.Ordinal);
            if (open < 0) return "";
            open += "<version>".Length;
            var close = text.IndexOf("</version>", open, StringComparison.Ordinal);
            if (close < 0) return "";
            return text[open..close].Trim();
        }
        catch
        {
            return "";
        }
    }
}
```

```csharp
public static class EnvCommand

    public static int Run(string[] args)
    {
        var json = args.Contains("--json");
        GameEnv env;
        try
        {
            env = GameEnvProbe.Discover();

            Console.Error.WriteLine("RD1001: " + ex.Message);
            return 4;

        if (json)
        {
            Console.WriteLine(
                System.Text.Json.JsonSerializer.Serialize(new
                {
                    gameRoot = env.GameRoot,
                    managedDir = env.ManagedDir,
                    managedDllCount = env.ManagedDllCount,
                    modsDir = env.ModsDir,
                    dataDir = env.DataDir,
                    workshopDir = env.WorkshopDir,
                    workshopExists = env.WorkshopExists,
                    configDir = env.ConfigDir,
                    modsConfigPath = env.ModsConfigPath,
                    playerLogPath = env.PlayerLogPath,
                    prevPlayerLogPath = env.PrevPlayerLogPath,
                    prevPlayerLogExists = env.PrevPlayerLogExists,
                    versionRaw = env.VersionRaw,
                    versionConfig = env.VersionConfig,
                    targetVersion = env.TargetVersion
                },
                new System.Text.Json.JsonSerializerOptions { WriteIndented = true }));
```

Create `_rimdev\engine\RimDev.Core\Metadata\MemberInfoIndex.cs`：

```csharp
using System.Reflection;

namespace RimDev.Core.Metadata;

public enum MemberKind { Field, Property, Method, Constructor, Type }

/// <summary>单个成员的存在性结论。</summary>
public sealed record MemberHit(
    MemberKind Kind,
    string Name,
    string DeclaringType,
    string OwnerType,
    string TypeName,
    bool IsPublic,
    bool IsStatic,
    string? Detail);

/// <summary>
/// 一个类型的成员表。
///
/// ★ 关键设计（由实测得出，见实施计划 §0.1）：
///   1. 字段必须包含 NonPublic —— 证据：VerbProperties.forcedMissRadius 是 private 字段，
///      却是本项目炮塔 XML 正在使用的合法字段。只查 public 会把它误报成"不存在"。
///   2. 查找大小写敏感 —— 证据：RaceProperties.IsMechanoid 存在，isMechanoid 不存在。
///   3. 字段与属性分开暴露 —— XML 只从字段读；规则包需要属性（如 IsMechanoid）。
/// </summary>
public sealed class MemberInfoIndex
{
    private const BindingFlags FieldFlags =
        BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Instance | BindingFlags.Static;

    // ★ 方法枚举必须一次 GetMethods、不能逐层 DeclaredOnly 累加：
    //   实测 Pawn.PreApplyDamage 逐层累加得到 3 个（含被覆盖的基类版本），
    //   而单次 GetMethods 得到 1 个（只返回最派生那个）—— 后者才是调用者要的语义。
    private const BindingFlags MethodFlags =
        BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Instance | BindingFlags.Static;

    private readonly Type _type;
    private readonly Dictionary<string, FieldInfo> _fields;
    private readonly Dictionary<string, PropertyInfo> _properties;
    private readonly Dictionary<string, List<MethodInfo>> _methods;

    public MemberInfoIndex(Type type)
    {
        _type = type;
        _fields = new Dictionary<string, FieldInfo>(StringComparer.Ordinal);
        _properties = new Dictionary<string, PropertyInfo>(StringComparer.Ordinal);
        _methods = new Dictionary<string, List<MethodInfo>>(StringComparer.Ordinal);

        foreach (var f in type.GetFields(FieldFlags))
            _fields.TryAdd(f.Name, f);

        foreach (var p in type.GetProperties(FieldFlags))
            _properties.TryAdd(p.Name, p);

        foreach (var m in type.GetMethods(MethodFlags))
        {
            if (!_methods.TryGetValue(m.Name, out var list))
                _methods[m.Name] = list = new List<MethodInfo>();
            list.Add(m);
        }
    }

    public Type Type => _type;
    public string FullName => _type.FullName ?? _type.Name;
    public string SimpleName => _type.Name;
    public int PublicFieldCount => _type.GetFields(BindingFlags.Public | BindingFlags.Instance).Length;
    public int AllFieldCount => _type.GetFields(FieldFlags).Length;
    public int PropertyCount => _type.GetProperties(BindingFlags.Public | BindingFlags.Instance).Length;
    public IReadOnlyCollection<string> FieldNames => _fields.Keys;
    public IReadOnlyCollection<string> PropertyNames => _properties.Keys;

    public IEnumerable<MemberHit> AllMembers()
    {
        foreach (var f in _fields.Values)
            yield return new MemberHit(MemberKind.Field, f.Name, f.DeclaringType?.Name ?? "?", _type.Name,
                f.FieldType.Name, f.IsPublic, f.IsStatic,
                f.IsInitOnly ? "readonly" : (f.IsPublic ? null : "non-public"));

        foreach (var p in _properties.Values)
            yield return new MemberHit(MemberKind.Property, p.Name, p.DeclaringType?.Name ?? "?", _type.Name,
                p.PropertyType.Name, p.GetMethod?.IsPublic ?? false,
                (p.GetMethod ?? p.SetMethod)?.IsStatic ?? false,
                "canWrite=" + p.CanWrite);

        foreach (var (name, list) in _methods)
            foreach (var m in list)
                yield return new MemberHit(MemberKind.Method, name, m.DeclaringType?.Name ?? "?", _type.Name,
                    m.ReturnType.Name, m.IsPublic, m.IsStatic, DescribeSignature(m));
    }

    public static string DescribeSignature(MethodInfo m)
    {
        var ps = m.GetParameters().Select(p =>
            (p.ParameterType.IsByRef ? "ref " : "") +
            p.ParameterType.Name.TrimEnd('&') + " " + p.Name);
        var mods = (m.IsStatic ? "static " : "") + (m.IsVirtual ? "virtual " : "") + (m.IsAbstract ? "abstract " : "");
        return $"{mods}{m.ReturnType.Name} {m.Name}({string.Join(", ", ps)})";
    }

    public FieldInfo? FindField(string name) => _fields.GetValueOrDefault(name);
    public PropertyInfo? FindProperty(string name) => _properties.GetValueOrDefault(name);
    public IReadOnlyList<MethodInfo> FindMethods(string name) =>
        _methods.TryGetValue(name, out var l) ? l : Array.Empty<MethodInfo>();

    /// <summary>按名字找成员，返回字段优先于属性（XML 只从字段读）。</summary>
    public MemberHit? FindMember(string name)
    {
        if (_fields.TryGetValue(name, out var f))
            return new MemberHit(MemberKind.Field, f.Name, f.DeclaringType?.Name ?? "?", _type.Name,
                f.FieldType.Name, f.IsPublic, f.IsStatic, f.IsInitOnly ? "readonly" : (f.IsPublic ? null : "non-public"));

        if (_properties.TryGetValue(name, out var p))
            return new MemberHit(MemberKind.Property, p.Name, p.DeclaringType?.Name ?? "?", _type.Name,
                p.PropertyType.Name, p.GetMethod?.IsPublic ?? false,
                (p.GetMethod ?? p.SetMethod)?.IsStatic ?? false, "canWrite=" + p.CanWrite);

        var m = FindMethods(name);
        if (m.Count > 0)
            return new MemberHit(MemberKind.Method, m[0].Name, m[0].DeclaringType?.Name ?? "?", _type.Name,
                m[0].ReturnType.Name, m[0].IsPublic, m[0].IsStatic, DescribeSignature(m[0]) + $"  (共 {m.Count} 个重载)");

        return null;
    }

    /// <summary>
    /// 给出最相近的成员名。用于把"字段不存在"变成"你是不是想写 X"。
    /// 优先级：忽略大小写全等 → 前缀 → 编辑距离。
    /// </summary>
    public IReadOnlyList<string> Suggest(string name, int max = 5)
    {
        var pool = _fields.Keys.Concat(_properties.Keys).Distinct().ToList();
        return pool
            .Select(n => (Name: n, Score: Similarity(name, n)))
            .Where(x => x.Score > 0)
            .OrderByDescending(x => x.Score)
            .ThenBy(x => x.Name.Length)
            .Take(max)
            .Select(x => x.Name)
            .ToList();
    }

    /// <summary>粗粒度相似度：越大越像；≤0 表示不像。</summary>
    public static double Similarity(string a, string b)
    {
        if (a.Length == 0 || b.Length == 0) return 0;
        if (string.Equals(a, b, StringComparison.OrdinalIgnoreCase))
            return string.Equals(a, b, StringComparison.Ordinal) ? 1000 : 900;   // 仅大小写差异最像

        var al = a.ToLowerInvariant();
        var bl = b.ToLowerInvariant();
        if (bl.StartsWith(al, StringComparison.Ordinal) || al.StartsWith(bl, StringComparison.Ordinal))
            return 500 - Math.Abs(a.Length - b.Length);

        var d = Levenshtein(al, bl);
        var maxLen = Math.Max(al.Length, bl.Length);
        var similarity = 1.0 - (double)d / maxLen;
        return similarity >= 0.6 ? similarity * 100 : 0;   // 太不像的不报，避免噪音
    }

    public static int Levenshtein(string a, string b)
    {
        var prev = new int[b.Length + 1];
        var cur = new int[b.Length + 1];
        for (var j = 0; j <= b.Length; j++) prev[j] = j;
        for (var i = 1; i <= a.Length; i++)
        {
            cur[0] = i;
            for (var j = 1; j <= b.Length; j++)
            {
                var cost = a[i - 1] == b[j - 1] ? 0 : 1;
                cur[j] = Math.Min(Math.Min(cur[j - 1] + 1, prev[j] + 1), prev[j - 1] + cost);
            }
            (prev, cur) = (cur, prev);
        }
        return prev[b.Length];
    }
}
```

```csharp
using System.Runtime.Loader;

namespace RimDev.Core.Metadata;

public sealed class ManagedAssemblyLoadException : Exception
{
    public ManagedAssemblyLoadException(string message, Exception inner) : base(message, inner) { }

/// <summary>
/// 把 RimWorld 的托管程序集载入当前进程并建立类型查找表。
///
/// 实测（实施计划 §0）：.NET 8 进程内 Assembly.LoadFrom(Assembly-CSharp.dll) 可行——
/// 12 ms 载入、26 个引用、GetTypes() 直接返回 16,157 个类型、只需解析 8 个依赖。
/// 因此 v1 不需要子进程隔离；但所有反射调用仍必须包 try/catch（某些类型元数据会抛 TypeLoadException）。
///
/// ★ 命名空间不可假设：BuildingProperties 在 RimWorld.*，ThingDef 在 Verse.*。
///   一律按简单名查找。
/// </summary>
public sealed class ManagedAssemblyCatalog
{
    private readonly Dictionary<string, List<Type>> _bySimpleName = new(StringComparer.Ordinal);
    private readonly Dictionary<string, MemberInfoIndex> _memberCache = new(StringComparer.Ordinal);
    private readonly Type[] _types;

    private ManagedAssemblyCatalog(string managedDir, Type[] types,
        IReadOnlyList<string> resolved, IReadOnlyList<string> resolveFailures,
        long loadMs, string? loaderWarning)
    {
        ManagedDir = managedDir;
        _types = types;
        ResolvedFromManaged = resolved;
        ResolveFailures = resolveFailures;
        LoadMilliseconds = loadMs;
        LoaderWarning = loaderWarning;

        foreach (var t in types)
        {
            var name = t.Name;
            if (!_bySimpleName.TryGetValue(name, out var list))
                _bySimpleName[name] = list = new List<Type>();
            list.Add(t);
        }
    }

    public string ManagedDir { get; }
    public int TypeCount => _types.Length;
    public IReadOnlyList<string> ResolvedFromManaged { get; }
    public IReadOnlyList<string> ResolveFailures { get; }
    public long LoadMilliseconds { get; }
    public string? LoaderWarning { get; }
    public IReadOnlyList<Type> AllTypes => _types;

    public static ManagedAssemblyCatalog Load(string managedDir, string assemblyName = "Assembly-CSharp.dll")
    {
        if (!Directory.Exists(managedDir))
            throw new ManagedAssemblyLoadException($"Managed 目录不存在: {managedDir}", new DirectoryNotFoundException(managedDir));

        var resolved = new List<string>();
        var failures = new List<string>();

        AssemblyLoadContext.Default.Resolving += (ctx, name) =>
        {
            var path = Path.Combine(managedDir, name.Name + ".dll");
            if (!File.Exists(path))
            {
                failures.Add(name.Name ?? "?");
                return null;
            }
            resolved.Add(name.Name ?? "?");
            return ctx.LoadFromAssemblyPath(path);
        };

        var sw = System.Diagnostics.Stopwatch.StartNew();
        Assembly asm;
        var asmPath = Path.Combine(managedDir, assemblyName);
        try
        {
            asm = Assembly.LoadFrom(asmPath);
        }
        catch (Exception ex)
        {
            // 实测这条路径不会触发；保留以便版本变化时能给出可操作报错
            throw new ManagedAssemblyLoadException(
                $"载入 {assemblyName} 失败（{ex.GetType().Name}）: {ex.Message}", ex);
        }

        string? loaderWarning = null;
        Type[] types;
        try
        {
            types = asm.GetTypes();
        }
        catch (ReflectionTypeLoadException ex)
        {
            // 历史教训：GetTypes() 只抛一次聚合异常，而 GetType(name) 会静默返回 null，
            // 极易被误判成"类型不存在"。这里显式降级并留下告警。
            types = ex.Types.Where(t => t is not null).Cast<Type>().ToArray();
            loaderWarning = $"ReflectionTypeLoadException：返回 {types.Length} 个类型，" +
                            $"加载器异常 {ex.LoaderExceptions.Length} 条。首条：{ex.LoaderExceptions.FirstOrDefault()?.Message}";
        }
        sw.Stop();

        return new ManagedAssemblyCatalog(managedDir, types, resolved, failures,
            sw.ElapsedMilliseconds, loaderWarning);
    }

    /// <summary>按简单名精确查找（不假设命名空间）。命中多个时返回第一个。</summary>
    public Type? FindBySimpleName(string simpleName) =>
        _bySimpleName.TryGetValue(simpleName, out var list) && list.Count > 0 ? list[0] : null;

    public IReadOnlyList<Type> FindAllBySimpleName(string simpleName) =>
        _bySimpleName.TryGetValue(simpleName, out var list) ? list : Array.Empty<Type>();

    /// <summary>支持 "Verse.ThingDef" 全名与 "ThingDef" 简单名两种写法。</summary>
    public Type? Find(string typeName)
    {
        var simple = typeName.Contains('.') ? typeName[(typeName.LastIndexOf('.') + 1)..] : typeName;
        if (_bySimpleName.TryGetValue(simple, out var list))
        {
            if (list.Count == 1) return list[0];
            var exact = list.FirstOrDefault(t => t.FullName == typeName);
            return exact ?? list[0];
        }
        return null;
    }

    public MemberInfoIndex MembersOf(Type type) =>
        _memberCache.TryGetValue(type.FullName ?? type.Name, out var idx)
            ? idx
            : _memberCache[type.FullName ?? type.Name] = new MemberInfoIndex(type);

    public MemberInfoIndex? MembersOf(string typeName)
    {
        var t = Find(typeName);
        return t is null ? null : MembersOf(t);
    }

    /// <summary>类型名字的候选建议（用于"类型不存在"时给相近项）。</summary>
    public IReadOnlyList<string> SuggestTypeNames(string typeName, int max = 5) =>
        _bySimpleName.Keys
            .Select(n => (Name: n, Score: MemberInfoIndex.Similarity(typeName, n)))
            .Where(x => x.Score > 0)
            .OrderByDescending(x => x.Score)
            .ThenBy(x => x.Name.Length)
            .Take(max)
            .Select(x => x.Name)
            .ToList();

    /// <summary>
    /// 判断某类型是否带"自定义 XML 加载器"（实例方法 LoadDataFromXmlCustom(XmlNode)）。
    /// 实测：全程序集 65 个类型带这个实例方法，静态方法 0 个 —— 所以必须查实例，不能查静态。
    /// 这是 shape 顾问（F2）判定"该写裸元素还是 &lt;li&gt;"的唯一依据。
    /// </summary>
    public static bool HasCustomXmlLoader(Type type)
    {
        try
        {
            var m = type.GetMethod("LoadDataFromXmlCustom",
                BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Instance);
            return m is not null && m.GetParameters().Length == 1;
        }
        catch
        {
            return false;
        }
    }
}
```

```csharp
using System.Text.Json;
using RimDev.Core.Diagnostics;
using RimDev.Core.Environment;
using RimDev.Core.Metadata;

namespace RimDev.Cli.Commands;

public static class ApiCommand
{
    private static readonly JsonSerializerOptions Json = new() { WriteIndented = true };

    public static int Run(string[] args)
    {
        // rimdev api <probe|type|member|ctor|list> <TypeName> [--member <name>] [--filter <sub>] [--json]
        if (args.Length < 2)
        {
            Console.Error.WriteLine("usage: rimdev api <probe|type|member|ctor|list> <TypeName> [--member x] [--filter s] [--json]");
            return 3;
        }

        var sub = args[1];
        var typeName = args.Length > 2 && !args[2].StartsWith("--") ? args[2] : null;
        var member = Option(args, "--member");
        var filter = Option(args, "--filter");
        var json = args.Contains("--json");

        if (typeName is null)
        {
            Console.Error.WriteLine("缺少类型名");
            return 3;
        }

        ManagedAssemblyCatalog cat;
        try
        {
            cat = ManagedAssemblyCatalog.Load(GameEnvProbe.Discover().ManagedDir);
        }
        catch (Exception ex)
        {
            Console.Error.WriteLine($"{Codes.AssemblyLoadFailed}: {ex.Message}");
            return 4;
        }

        var idx = cat.MembersOf(typeName);
        if (idx is null)
        {
            var diag = new Diagnostic(Codes.TypeNotFound, Severity.Error, Scope.Offline,
                $"类型 {typeName} 不存在", "用 rimdev api list --filter <片段> 找相近类型",
                TypeName: typeName, Suggestions: cat.SuggestTypeNames(typeName, 6));
            return Emit(json, diag, null);
        }

        return sub switch
        {
            "probe" => ProbeMember(json, idx, member),
            "type" => TypeSummary(json, idx, filter),
            "member" => ProbeMember(json, idx, member),
            "ctor" => Ctors(json, idx),
            "list" => ListTypes(json, cat, filter),
            _ => Unknown(sub)
        };
    }

    private static int ProbeMember(bool json, MemberInfoIndex idx, string? member)
    {
        if (member is null)
        {
            // 无成员名 → 只报类型存在
            var info = new Diagnostic(Codes.MetadataUnavailable, Severity.Info, Scope.Offline,
                $"类型 {idx.FullName} 存在", TypeName: idx.FullName,
                Evidence: new Dictionary<string, string>
                {
                    ["publicFields"] = idx.PublicFieldCount.ToString(),
                    ["allFields"] = idx.AllFieldCount.ToString(),
                    ["properties"] = idx.PropertyCount.ToString(),
                    ["baseType"] = idx.Type.BaseType?.Name ?? ""
                });
            return Emit(json, info, idx.FullName);
        }

        var hit = idx.FindMember(member);
        if (hit is null)
        {
            var suggestions = idx.Suggest(member, 6);
            // 若候选里存在仅大小写不同的项，按 RD4005 报（这是历史真实错误类型）
            var caseOnly = suggestions.FirstOrDefault(s => string.Equals(s, member, StringComparison.OrdinalIgnoreCase));
            var code = caseOnly is not null ? Codes.CaseMismatch : Codes.FieldNotFound;
            var diag = new Diagnostic(code, Severity.Error, Scope.Offline,
                $"{idx.FullName} 上没有成员 {member}", "用建议名替换",
                TypeName: idx.FullName, MemberName: member, Suggestions: suggestions);
            return Emit(json, diag, idx.FullName);
        }

        var detail = new Dictionary<string, string>
        {
            ["kind"] = hit.Kind.ToString(),
            ["type"] = hit.TypeName,
            ["declaredBy"] = hit.DeclaringType,
            ["isPublic"] = hit.IsPublic.ToString(),
            ["isStatic"] = hit.IsStatic.ToString()
        };
        if (hit.Detail is not null) detail["detail"] = hit.Detail;

        // 字段是否带自定义 XML 加载器（决定 XML 该写裸元素还是 <li>）
        if (hit.Kind == MemberKind.Field)
        {
            var f = idx.FindField(member);
            if (f is not null)
            {
                var elem = Nullable.GetUnderlyingType(f.FieldType) ?? f.FieldType;
                detail["elementType"] = elem.Name;
                detail["customXmlLoader"] = ManagedAssemblyCatalog.HasCustomXmlLoader(elem).ToString();
                detail["elementIsList"] = elem.IsGenericType &&
                    elem.GetGenericTypeDefinition() == typeof(List<>);
                if (detail["elementIsList"] == "True")
                    detail["listElementType"] = elem.GetGenericArguments()[0].Name;
            }
        }

        var ok = new Diagnostic(Codes.MetadataUnavailable, Severity.Info, Scope.Offline,
            $"{idx.FullName}.{hit.Name} 存在（{hit.Kind}）", TypeName: idx.FullName, MemberName: hit.Name,
            Evidence: detail);
        return Emit(json, ok, idx.FullName);
    }

                diagnostics = new[]

                    new
                    {
                        code = diag.Code,
                        severity = diag.Severity.ToString().ToLowerInvariant(),
                        scope = diag.Scope.ToString().ToLowerInvariant(),
                        message = diag.Message,
                        hint = diag.Hint,
                        type = diag.TypeName,
                        member = diag.MemberName,
                        suggestions = diag.Suggestions,
                        evidence = diag.Evidence,
                        fingerprint = diag.Fingerprint
                    }
                }
            }, Json));
        }
        else
        {
            var prefix = diag.Severity == Severity.Error ? "✗" : "✓";
            Console.WriteLine($"{prefix} {diag.Message}");
            if (diag.Hint is not null) Console.WriteLine($"  提示: {diag.Hint}");
            if (diag.Suggestions is { Count: > 0 })
                Console.WriteLine($"  候选: {string.Join(", ", diag.Suggestions)}");
            if (diag.Evidence is not null)
                foreach (var (k, v) in diag.Evidence)

        return diag.Severity == Severity.Error ? 2 : 0;

        return 3;
```

```csharp
using RimDev.Core.Diagnostics;

namespace RimDev.Core.Metadata;

public enum NameKind { Type, Field, Property, Method, EnumMember, DefName, Unknown }

public sealed record NameVerdict(Diagnostic Diagnostic, NameKind Kind, bool Exists);

/// <summary>
/// 名字查证器（设计文档 02 §3-F1 的 v1 子集）。
/// 动机：本项目因凭语义猜"原版名字"造成 8 次加载期错误
/// （Repair→Construction、CarryCapacity→CarryingCapacity、MechanoidCentipede→MechanicalCentipede、
///  Shot_MachineGun/Building_Collapse 不存在、canBeDeconstructed 等三字段不存在）。
/// 本类把这些查证从"开游戏才知道"提前到命令行。
/// </summary>
public sealed class NameOracle
{
    private readonly ManagedAssemblyCatalog _cat;

    public NameOracle(ManagedAssemblyCatalog cat) => _cat = cat;

    public NameVerdict Check(string name, NameKind kind = NameKind.Unknown, string? ownerType = null)
    {
        return kind switch
        {
            NameKind.Type => CheckType(name),
            NameKind.Field or NameKind.Property or NameKind.Method => CheckMember(name, ownerType, kind),
            NameKind.EnumMember => CheckEnumMember(name, ownerType),
            NameKind.DefName => NotImplementedDefName(name),
            _ => CheckAny(name)
        };
    }

    private NameVerdict CheckType(string name)
    {
        var t = _cat.Find(name);
        if (t is not null)
            return new NameVerdict(new Diagnostic(Codes.MetadataUnavailable, Severity.Info, Scope.Offline,
                $"类型 {name} 存在 → {t.FullName}", TypeName: t.FullName), NameKind.Type, true);

        return new NameVerdict(new Diagnostic(Codes.TypeNotFound, Severity.Error, Scope.Offline,
            $"类型 {name} 不存在", "用候选中的全名替换；注意命名空间不可假设（BuildingProperties 在 RimWorld 而非 Verse）",
            TypeName: name, Suggestions: _cat.SuggestTypeNames(name, 6)), NameKind.Type, false);
    }

    private NameVerdict CheckMember(string name, string? ownerType, NameKind kind)
    {
        if (ownerType is null)
            return SearchMemberAcrossTypes(name, kind);

        var idx = _cat.MembersOf(ownerType);
        if (idx is null)
            return CheckType(ownerType);

        var hit = idx.FindMember(name);
        if (hit is not null)
            return new NameVerdict(new Diagnostic(Codes.MetadataUnavailable, Severity.Info, Scope.Offline,
                $"{idx.FullName}.{name} 存在（{hit.Kind} {hit.TypeName}）", TypeName: idx.FullName, MemberName: name),
                hit.Kind, true);

        var suggestions = idx.Suggest(name, 6);
        var caseOnly = suggestions.FirstOrDefault(s => string.Equals(s, name, StringComparison.OrdinalIgnoreCase));
        return new NameVerdict(new Diagnostic(
            caseOnly is not null ? Codes.CaseMismatch : Codes.FieldNotFound,
            Severity.Error, Scope.Offline,
            $"{idx.FullName} 上没有成员 {name}",
            caseOnly is not null ? $"大小写不符，正确写法是 {caseOnly}" : "用候选名替换",
            TypeName: idx.FullName, MemberName: name, Suggestions: suggestions),
            kind, false);
    }

    private NameVerdict CheckEnumMember(string name, string? ownerType)
    {
        if (ownerType is null)
        {
            foreach (var t in _cat.AllTypes.Where(t => t.IsEnum))
            {
                if (Enum.GetNames(t).Contains(name))
                    return new NameVerdict(new Diagnostic(Codes.MetadataUnavailable, Severity.Info, Scope.Offline,
                        $"枚举成员 {name} 存在 → {t.FullName}.{name}", TypeName: t.FullName, MemberName: name),
                        NameKind.EnumMember, true);
            }
            return new NameVerdict(new Diagnostic(Codes.EnumValueInvalid, Severity.Error, Scope.Offline,
                $"找不到枚举成员 {name}", "指定 --owner <EnumType> 可收窄搜索"),
                NameKind.EnumMember, false);
        }

        var et = _cat.Find(ownerType);
        if (et is null || !et.IsEnum)
            return new NameVerdict(new Diagnostic(Codes.TypeNotFound, Severity.Error, Scope.Offline,
                $"{ownerType} 不是已加载的枚举类型", TypeName: ownerType), NameKind.EnumMember, false);

        var names = Enum.GetNames(et);
        if (names.Contains(name))
            return new NameVerdict(new Diagnostic(Codes.MetadataUnavailable, Severity.Info, Scope.Offline,
                $"{et.FullName}.{name} 存在", TypeName: et.FullName, MemberName: name), NameKind.EnumMember, true);

        return new NameVerdict(new Diagnostic(Codes.EnumValueInvalid, Severity.Error, Scope.Offline,
            $"{et.FullName} 上没有枚举成员 {name}", "用合法值之一替换", TypeName: et.FullName, MemberName: name,
            Suggestions: names.Where(n => MemberInfoIndex.Similarity(name, n) > 0).Take(6).ToList()),
            NameKind.EnumMember, false);
    }

    /// <summary>未指定宿主类型时跨全部类型搜索成员名（16157 个类型，耗时约百毫秒级）。</summary>
    private NameVerdict SearchMemberAcrossTypes(string name, NameKind kind)
    {
        var hits = new List<string>();
        foreach (var t in _cat.AllTypes)
        {
            try
            {
                const System.Reflection.BindingFlags f =
                    System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.NonPublic |
                    System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.Static;

                var fi = t.GetField(name, f);
                if (fi is not null) { hits.Add($"{t.FullName} (Field {fi.FieldType.Name})"); if (hits.Count >= 12) break; continue; }

                var pi = t.GetProperty(name, f);
                if (pi is not null) { hits.Add($"{t.FullName} (Property {pi.PropertyType.Name})"); if (hits.Count >= 12) break; }
            }
            catch
            {
                // 个别类型元数据不可读属正常
            }
        }

        if (hits.Count > 0)
            return new NameVerdict(new Diagnostic(Codes.MetadataUnavailable, Severity.Info, Scope.Offline,
                $"成员名 {name} 在 {hits.Count} 个类型上存在", "用 --owner <Type> 收窄", MemberName: name,
                Suggestions: hits), kind, true);

        return new NameVerdict(new Diagnostic(Codes.FieldNotFound, Severity.Error, Scope.Offline,
            $"全程序集找不到名为 {name} 的成员", "确认拼写；或用 rimdev api list --filter 找类型",
            MemberName: name), kind, false);
    }

    private NameVerdict CheckAny(string name)
    {
        var t = _cat.Find(name);
        if (t is not null)
            return new NameVerdict(new Diagnostic(Codes.MetadataUnavailable, Severity.Info, Scope.Offline,
                $"{name} 是一个类型 → {t.FullName}", TypeName: t.FullName), NameKind.Type, true);

        var memberVerdict = SearchMemberAcrossTypes(name, NameKind.Unknown);
        if (memberVerdict.Exists) return memberVerdict;

        var suggestions = _cat.SuggestTypeNames(name, 6);
        return new NameVerdict(new Diagnostic(Codes.TypeNotFound, Severity.Error, Scope.Offline,
            $"{name} 既不是类型也不是成员",
            "v1 只查类型/字段/属性/方法/枚举；Def 名的查证需要 Def 索引（v2）",
            TypeName: name, MemberName: name, Suggestions: suggestions), NameKind.Unknown, false);
    }

    /// <summary>Def 名查证需要 v2 的 Def 索引，v1 明确不做，且必须如实说明而不是猜。</summary>
    private static NameVerdict NotImplementedDefName(string name) =>
        new(new Diagnostic(Codes.MetadataUnavailable, Severity.Warning, Scope.Offline,
            $"Def 名 {name} 的存在性检查尚未实现（需要 Def 索引，属 v2 范围）",
            "v1 已支持 --kind type|field|property|method|enum；Def 名请先用 rimsearcher list --type <DefType>",
            TypeName: name, MemberName: name), NameKind.DefName, false);
}
```

```csharp
                command = "name check",
                query = new { name, kind = kind.ToString(), owner },
                exists = verdict.Exists,
                resolvedKind = verdict.Kind.ToString(),
                diagnostics = new[]
                {
                    new
                    {
                        code = d.Code,
                        severity = d.Severity.ToString().ToLowerInvariant(),
                        scope = d.Scope.ToString().ToLowerInvariant(),
                        message = d.Message,
                        hint = d.Hint,
                        suggestions = d.Suggestions,
                        fingerprint = d.Fingerprint
                    }
                }
            }, Json));
        }
        else
        {
            Console.WriteLine((verdict.Exists ? "✓ 存在  " : "✗ 不存在 ") + d.Message);
            if (d.Hint is not null) Console.WriteLine("  提示: " + d.Hint);
            if (d.Suggestions is { Count: > 0 })
            {
                Console.WriteLine(verdict.Exists && verdict.Kind == NameKind.Unknown ? "  命中:" : "  候选:");
                foreach (var s in d.Suggestions) Console.WriteLine("    - " + s);
            }
        }

        return d.Severity == Severity.Error ? 2 : 0;
    }

    private static NameKind ParseKind(string? s) => s?.ToLowerInvariant() switch
    {
        "type" => NameKind.Type,
        "field" => NameKind.Field,
        "property" or "prop" => NameKind.Property,
        "method" => NameKind.Method,
        "enum" => NameKind.EnumMember,
        "def" => NameKind.DefName,
        _ => NameKind.Unknown
    };

    private static string? Option(string[] args, string name)
    {
        var i = Array.IndexOf(args, name);
        return i >= 0 && i + 1 < args.Length ? args[i + 1] : null;
    }
}
```

````markdown
## 这是什么

离线的环世界 Mod 开发辅助工具。v1 覆盖三件事：
````
